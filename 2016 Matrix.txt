import java.util.*;
public class SqMat
{
    public static void main(String args[])
    {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter the size of the array :");
        int M = sc.nextInt();
        if(M>3 && M<10)
        {
            System.out.println("Valid Input");
        }
        else
        {
            System.out.println("INvalid Input");
            System.exit(0);
        }
        int A[][] = new int[M][M];
        int b[] = new int[M];
        int i=0,j=0;
        for(i=0;i<M;i++)        // input function
        {
            for(j=0;j<M;j++)
            {
                System.out.println("Enter a Number");
                A[i][j] = sc.nextInt();
            }
        }
        System.out.println("ORIGINAL MATRIX :");
        for(i=0;i<M;i++)        // displaying original matrix
        {
            for(j=0;j<M;j++)
            {
                System.out.print(A[i][j] + " ");
            }
            System.out.println();
        }
        int k=0;
        for(i=0;i<M;i++)        // transferring non-boundary elements
        {
            for(j=0;j<M;j++)
            {
               if(i!=0 && i!=M-1 && j!=0 && j!=M-1)
                {
                    b[k] = A[i][j];
                    k++;
                } 
            }
        }
        for(i=0;i<b.length-1;i++)       // sorting
        {
            for(j=0;j<b.length-1-i;j++)
            {
                if(b[j] > b[j+1])
                {
                    int t = b[j];
                    b[j] = b[j+1];
                    b[j+1] = t;
                }
            }
        }
        k=0;
        for(i=0;i<M;i++)        // transferring sorted elements
        {
            for(j=0;j<M;j++)
            {
               if(i!=0 && i!=M-1 && j!=0 && j!=M-1)
                {
                    A[i][j] = b[k];
                    k++;
                } 
            }
        }
        int sum = 0;
        for(i=0;i<M;i++)        // calculating sum of diagonal elements
        {
            for(j=0;j<M;j++)
            {
                if(i == j || i + j == M-1)
                {
                    sum = sum + A[i][j];
                }
            }
        }
        System.out.println("RE - ARRANGED MATRIX");
        for(i=0;i<M;i++)        // displaying re-arranged matrix
        {
            for(j=0;j<M;j++)
            {
                System.out.print(A[i][j] + " ");
            }
            System.out.println();
        }
        System.out.println("DIAGONAL ELEMENTS");
        for(i=0;i<M;i++)        // displaying diagonal elements
        {
            for(j=0;j<M;j++)
            {
                if(i == j || i + j == M-1)
                {
                    System.out.print(A[i][j]);
                }
                else
                {
                    System.out.print(" ");
                }
            }
            System.out.println();
        }
        System.out.println("SUM OF DIAGONAL ELEMENTS = " + sum);
    }
}
