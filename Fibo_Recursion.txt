import java.util.*;
public class Recursion
{
    int a;
    int b;
    int c;
    int limit;
    
    public Recursion()
    {
        a=0;
        b=0;
        c=0;
        limit=0;
    }
    
    void input()
    {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter the limit of the series");
        limit=sc.nextInt();
    }
    
    int fib(int n)
    {
        if(n==1)
        {
            return(0);
        }
        if(n==2)
        {
            return(1);
        }
        
        return(fib(n-1) + fib(n-2));
    }
    
    void generate_fibseries()
    {
        int i=0;
        for(i=1;i<=limit;i++)
        {
            int t = fib(i);
            System.out.print(t + " ");
        }
    }
    public static void main(String args[])
    {
        Recursion ob = new Recursion();
        ob.input();
        ob.generate_fibseries();
    }

}
