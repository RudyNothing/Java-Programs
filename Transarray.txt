import java.util.*;
public class Transarray
{
    int arr[][];
    int m;
    int n;
    
    public Transarray()
    {
        m=0;
        n=0;
        arr = new int[m][n];
    }
    public Transarray(int mm,int nn)
    {
        m=mm;
        n=nn;
        if(m<20 && n<20)
        {
            System.out.println("Valid Input");
        }
        else
        {
            System.exit(0);
        }
        arr = new int[m][n];
    }
    void fillarray()
    {
        Scanner sc = new Scanner(System.in);
        int i=0,j=0;
        for(i=0;i<m;i++)
        {
            for(j=0;j<n;j++)
            {
                System.out.println("Enter a Number");
                arr[i][j] = sc.nextInt();
            }
        }
    }
    void transpose(Transarray A)
    {
        int i=0,j=0;
        for(i=0;i<A.m;i++)
        {
            for(j=0;j<A.n;j++)
            {
                arr[i][j] = A.arr[j][i];
            }
        }
    }
    void disparray()
    {
        int i=0,j=0;
        for(i=0;i<m;i++)
        {
            for(j=0;j<n;j++)
            {
                System.out.print(arr[i][j] + " ");
            }
            System.out.println();
        }
    }
    public static void main(String args[])
    {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter the number of rows");
        int mm = sc.nextInt();
        System.out.println("Enter the number of colums");
        int nn = sc.nextInt();
        if(mm <=20 && nn <=20)
        {
            System.out.println("Valid Input");
        }
        else
        {
            System.out.println("Wrong Input");
            System.exit(0);
        }
        Transarray ob = new Transarray(mm,nn);
        ob.fillarray();
        ob.disparray();
        Transarray ob1 = new Transarray(mm,nn);
        ob1.transpose(ob);
        System.out.println("The Transposed Matrix is :");
        ob1.disparray();
    }

}
