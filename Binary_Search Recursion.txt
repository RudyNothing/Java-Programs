import java.util.*;
public class Binary
{
    int A[] = new int[100];
    int n;
    int l;
    int u;
    
    public Binary(int nn)
    {
        n = nn;
        l = 0;
        u = n-1;
    }
    
    void readdata()
    {
        Scanner sc = new Scanner(System.in);
        int i=0;
        for(i=0;i<n;i++)
        {
            System.out.println("Enter a Number");
            A[i] = sc.nextInt();
        }
    }
    
    int binary_search(int v)
    {
        if(l>u)
        {
            return(-1);
        }
        else
        {
            int mid = (l + u)/2;
            if(v == A[mid])
            {
                return(mid +1);
            }
            else if(v > A[mid])
            {
                l = mid + 1;
            }
            else
            {
                u = mid-1;
            }
            return(binary_search(v));
        }
    }
    public static void main(String args[])
    {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter the size of the array");
        int nn = sc.nextInt();
        Binary ob = new Binary(nn);
        ob.readdata();
        System.out.println("Enter the number to be searched");
        int v = sc.nextInt();
        int b = ob.binary_search(v);
        if(b>0)
        {
            System.out.println("Found at " + b + " location.");
        }
        else if(b==-1)
        {
            System.out.println("Not Found");
        }
    }
}
