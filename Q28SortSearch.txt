import java.util.*;
public class Sort
{
    int arr[];
    int item;
    
    void inpdata()
    {
        arr= new int[50];
        int i=0;
        Scanner sc = new Scanner(System.in);
        for(i=0;i<50;i++)
        {
            System.out.println("Enter a Number(no duplicate numbers to be entered):");
            arr[i]= sc.nextInt();
        }
        
    }
    void bubsort()
    {
        int i=0,j=0,t=0;
        for(i=0;i<arr.length-1;i++) //Passes
        {
            for(j=0;j<arr.length-1-i;j++)  //comparisons
            {
                if(arr[j]>arr[j+1])
                {
                    t=arr[j];
                    arr[j]=arr[j+1];
                    arr[j+1]=t;
                }
            }
        }
        System.out.print("Sorted Array : ");
        for(i=0;i<arr.length;i++)
        {
            System.out.print(arr[i] + " ");
        }
        System.out.println();
    }
    void binsearch()
    {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter the Number to be found:");
        item=sc.nextInt();
        int low=0,high=arr.length-1,mid=0,found=0;
        while(low<=high)
        {
            mid=(low + high)/2;
            if(item==arr[mid])
            {
                found = 1;
                System.out.println("Found at " + (mid + 1) + " position.");
                break;
            }
            else if(item>arr[mid])
                low = mid + 1;
            else
                high = mid - 1;
        }
        if(found == 0)
            System.out.println("Not Found");
    }
    public static void main(String args[])
    {
        Sort ob = new Sort();
        ob.inpdata();
        ob.bubsort();
        ob.binsearch();
    }
}
