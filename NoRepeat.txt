import java.util.*;
public class NoRepeat
{
    String word;
    int len;
    
    public NoRepeat(String wd)
    {
        word=wd;
        len=word.length();
    }
    boolean check()
    {
        int i=0,c=0;
        for ( i = 0; i < len; i++) 
        {
            char ch = word.charAt(i);
                if (Character.isLetter(ch) && word.indexOf(ch, i + 1) != -1) 
                {
                    return(true);
        }       }
        return(false);
    }
    void prn()
    {
        boolean c = check();
        if(c==true)
        {
            System.out.println("The Word " + word + " has REPEATED ALPHABETS.");
        }
        else
        {
            System.out.println("The Word " + word + " has NO REPEATED ALPHABETS.");
        }
    }
    public static void main(String args[])
    {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter the word to be checked");
        String wd = sc.nextLine();
        NoRepeat ob = new NoRepeat(wd);
        ob.prn();
        
    }
}
